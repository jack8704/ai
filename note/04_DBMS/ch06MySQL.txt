	-- DCL(계정생성, 권한부여, 권한박탈, 계정삭제)
	     create user userid identified by 'password'; -- 계정 생성
	     grant all privileges on *.* to userid; -- 권한 부여
	     revoke all on *.* from userid; -- 권한 박탈
             drop user userid; -- 계정 삭제
